cmake_minimum_required(VERSION 3.16)
project(Co60Transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Geant4 11.4 REQUIRED)
include(${Geant4_USE_FILE})
find_package(ROOT CONFIG REQUIRED COMPONENTS Core RIO Tree)
add_executable(co60_transport transport.cc)
target_include_directories(co60_transport PRIVATE ../../integration/annie_ratpac)
target_link_libraries(co60_transport PRIVATE ${Geant4_LIBRARIES} ROOT::Core ROOT::RIO ROOT::Tree)
target_compile_options(co60_transport PRIVATE -Wall -Wextra -Wpedantic)
